            const threeDaysMs = 3 * 24 * 60 * 60 * 1000;

            // กรองสินค้า: ถ้ารถมอเตอร์ไซค์หมดสต็อก (stock <= 0) และผ่านไปนานกว่า 3 วันแล้ว จะซ่อนจากหน้า Index ทันที

    function addMergedCart(id, name, price, stock, imageUrl) {
        const qtyInput = document.getElementById(`qty-${id}`);
        const buyQty = parseInt(qtyInput.value) || 1;
        if (buyQty <= 0) { showToast('⚠️ กรุณาระบุจำนวนมากกว่า 0 ครับ'); return; }

        const existingItem = cart.find(item => item.id === id);
        const currentInCart = existingItem ? existingItem.quantity : 0;

        if ((currentInCart + buyQty) > stock) {
            showToast(`⚠️ สต็อกไม่พอ! สินค้า "${name}" ในคลังเหลือ ${stock} ชิ้น`);
            return;
        }

        if (existingItem) existingItem.quantity += buyQty;
        else cart.push({ id, name, price, imageUrl, quantity: buyQty });

        saveCartToServer();
        updateCartUI();
        showToast(`เพิ่ม "${name}" จำนวน ${buyQty} ชิ้นลงตะกร้าแล้ว`);
    }

    function changeCartQty(index, delta) {
        const item = cart[index];
        const prod = allProducts.find(p => p.id === item.id);
        const stockMax = prod ? prod.stock : 999;

        const newQty = item.quantity + delta;
        if (newQty <= 0) { removeFromCart(index); return; }
        if (newQty > stockMax) { showToast(`⚠️ สินค้า "${item.name}" ในคลังมีเหลือแค่ ${stockMax} ชิ้นครับ`); return; }

        item.quantity = newQty;
        saveCartToServer();
        updateCartUI();
    }

    function removeFromCart(index) {
        const removedName = cart[index].name;
        cart.splice(index, 1);
        saveCartToServer();
        updateCartUI();
        showToast(`ลบ "${removedName}" ออกจากตะกร้าแล้ว`);
    }

                            <button onclick="changeCartQty(${index}, 1)">+</button>
                        </div>
                        <span style="color:#ff4d4f; font-weight:500; min-width:55px; text-align:right;">฿${(item.price * item.quantity).toLocaleString()}</span>
                        <button class="btn-remove-item" onclick="removeFromCart(${index})"><i class="fa-solid fa-trash-can"></i></button>
                    </div>
                </div>`; 
            total += (parseFloat(item.price) * item.quantity); 
        });
        document.getElementById('totalPrice').innerText = total.toLocaleString();
        updateDynamicQR(total);
    }

    function updateDynamicQR(amount = 0) {
        document.getElementById('qrTotalDisplay').innerText = amount.toLocaleString();
        const qrWrapper = document.getElementById('dynamicQrWrapper');
        const promptPayNumber = storeSettings.promptpay_phone || '0812768242';

        if (amount <= 0) {
            qrWrapper.innerHTML = `<p style="color:#888; font-size:12px; padding:20px;">กรุณาเลือกสินค้าก่อนสแกนชำระเงิน</p>`;
            return;
        }

        const apiQrUrl = `https://promptpay.io/${promptPayNumber}/${amount}.png`;
        qrWrapper.innerHTML = `<img src="${apiQrUrl}" alt="PromptPay QR" style="width:190px; height:190px; object-fit:contain; display:block;">`;
    }

    async function checkout() {
        const name = document.getElementById('customerName').value.trim(); 
        const phone = document.getElementById('customerPhone').value.trim();
        const address = document.getElementById('customerAddress').value.trim();
        const slipInput = document.getElementById('slipFile');

        if (cart.length === 0) { showToast('⚠️ ตะกร้าว่างเปล่าครับ'); return; }
        if (!name || !phone || !address) { showToast('⚠️ กรุณากรอก ชื่อ, เบอร์โทร และ ที่อยู่จัดส่ง ให้ครบถ้วน'); return; }
        if (selectedPayment === 'qr' && slipInput.files.length === 0) { showToast('⚠️ กรุณาแนบสลิปโอนเงินเพื่อยืนยันคำสั่งซื้อครับ'); return; }
        
        const btn = document.querySelector('.checkout-btn'); 
        btn.innerText = 'กำลังส่งคำสั่งซื้อ...'; btn.disabled = true;
        
        try {
            let slipUrl = '';
            if (selectedPayment === 'qr' && slipInput.files.length > 0) {
                const file = slipInput.files[0];
                const fileName = `slip_${Date.now()}.${file.name.split('.').pop()}`;
                const { error: upErr } = await db.storage.from('product-images').upload(fileName, file);
                if (upErr) throw upErr;
                const { data: urlD } = db.storage.from('product-images').getPublicUrl(fileName);
                slipUrl = urlD.publicUrl;
            }

            for (let item of cart) {
                const prod = allProducts.find(p => p.id === item.id);
                if (!prod || prod.stock < item.quantity) {
                    throw new Error(`สินค้า "${item.name}" ในคลังเหลือไม่พอ`);
                }
            }

            const total = cart.reduce((sum, item) => sum + (parseFloat(item.price) * item.quantity), 0);
            const payText = selectedPayment === 'qr' ? `โอนจ่ายผ่าน QR พร้อมเพย์ (${storeSettings.promptpay_phone || '0812768242'})` : 'ติดต่อร้าน/จ่ายหน้าร้าน';

            const { data: oData, error: oError } = await db.from('orders').insert([{ 
                customer_name: name, customer_phone: phone, contact_info: phone, address: address,
                total_price: total, payment_method: payText, payment_status: selectedPayment === 'qr' ? 'รอตรวจสอบสลิป' : 'ชำระเงินหน้าร้าน',
                slip_url: slipUrl
            }]).select();
            
            if (oError) throw oError;
            
            const orderId = oData[0].id;
            const items = cart.map(item => ({ order_id: orderId, product_id: item.id, quantity: item.quantity, price: item.price }));
            await db.from('order_items').insert(items);

            for (let item of cart) {
                const prod = allProducts.find(p => p.id === item.id);
                // ตัดสต็อกสินค้า และบันทึกเวลา updated_at เป็นปัจจุบัน (ใช้อ้างอิงนับถอยหลัง 3 วันสำหรับรถมอเตอร์ไซค์ที่ขายแล้ว)
                await db.from('products').update({ stock: prod.stock - item.quantity, updated_at: new Date() }).eq('id', item.id);
            }

            for (let item of cart) {
                const prodInfo = allProducts.find(p => p.id === item.id);
                const cost = prodInfo ? (prodInfo.cost_price || 0) : 0;
                await db.from('sales_history').insert([{
                    product_name: item.name, quantity: item.quantity, buy_price: cost, sell_price: item.price, total_profit: ((item.price - cost) * item.quantity)
                }]);
            }

            cart = [];
            saveCartToServer();
            updateCartUI();
            toggleCart();
            showToast('🎉 สั่งซื้อสำเร็จเรียบร้อย!');
            fetchProducts();
        } catch (err) {
            showToast('⚠️ เกิดข้อผิดพลาด: ' + err.message);
        } finally {
            btn.innerText = 'ยืนยันการสั่งซื้อ'; btn.disabled = false;
        }
    }

    fetchProducts();
</script>
